(
    SELECT MAX(Book_Count)

Q49)
SELECT b.Author_Name, COUNT(*) AS Issue_Count
FROM Member m
JOIN Issue i ON m.Member_Id = i.Member_Id
JOIN Books b ON i.Book_No = b.Book_No
WHERE m.Membership_Type = 'Lifetime'
GROUP BY b.Author_Name
HAVING COUNT(*) =
(
    SELECT MIN(Issue_Count)
    FROM
    (
        SELECT COUNT(*) AS Issue_Count
        FROM Member m2
        JOIN Issue i2 ON m2.Member_Id = i2.Member_Id
        JOIN Books b2 ON i2.Book_No = b2.Book_No
        WHERE m2.Membership_Type = 'Lifetime'
        GROUP BY b2.Author_Name
    ) AS Counts
);
