MySQL Assignment – 4

Q1)
SELECT first_name AS First_Name, last_name AS Last_Name
FROM EMPLOYEES;

Q2)
SELECT DISTINCT department_id
FROM EMPLOYEES;

Q3)
SELECT *
FROM EMPLOYEES
ORDER BY first_name DESC;

Q4)
SELECT first_name, last_name, salary, salary * 0.15 AS PF
FROM EMPLOYEES;

Q5)
SELECT employee_id, CONCAT(first_name, ' ', last_name) AS employee_name, salary
FROM EMPLOYEES
ORDER BY salary ASC;

Q6)
SELECT SUM(salary) AS total_salary
FROM EMPLOYEES;

Q7)
SELECT MAX(salary) AS highest_salary, MIN(salary) AS lowest_salary
FROM EMPLOYEES;

Q8)
SELECT AVG(salary) AS average_salary, COUNT(*) AS total_employees
FROM EMPLOYEES;

Q9)
SELECT COUNT(*) AS total_employees
FROM EMPLOYEES;

Q10)
SELECT COUNT(DISTINCT job_id) AS different_job_roles
FROM EMPLOYEES;

Q11)
SELECT *
FROM EMPLOYEES
LIMIT 10;

Q12)
SELECT CONCAT(first_name, ' ', last_name) AS employee_name, salary
FROM EMPLOYEES
WHERE salary < 10000 OR salary > 15000;

Q13)
SELECT CONCAT(first_name, ' ', last_name) AS employee_name, department_id
FROM EMPLOYEES
WHERE department_id IN (30, 100)
ORDER BY department_id;

Q14)
SELECT CONCAT(first_name, ' ', last_name) AS employee_name, salary
FROM EMPLOYEES
WHERE department_id IN (30, 100)
AND salary NOT BETWEEN 10000 AND 15000;

Q15)
SELECT CONCAT(first_name, ' ', last_name) AS employee_name, hire_date
FROM EMPLOYEES
WHERE YEAR(hire_date) = 1987;

Q16)
SELECT first_name
FROM EMPLOYEES
WHERE LOWER(first_name) LIKE '%b%'
AND LOWER(first_name) LIKE '%c%';

Q17)
SELECT e.last_name, j.job_title AS job_role, e.salary
FROM EMPLOYEES e
JOIN JOBS j ON e.job_id = j.job_id
WHERE j.job_title IN ('Programmer', 'Shipping Clerk')
AND e.salary NOT IN (4500, 10000, 15000);

Q18)
SELECT last_name
FROM EMPLOYEES
WHERE CHAR_LENGTH(last_name) = 6;

Q19)
SELECT last_name
FROM EMPLOYEES
WHERE LOWER(last_name) LIKE '__e%';

Q20)
SELECT DISTINCT job_id
FROM EMPLOYEES;

Q21)
SELECT *
FROM EMPLOYEES
WHERE last_name IN ('BLAKE', 'SCOTT', 'KING', 'FORD');
